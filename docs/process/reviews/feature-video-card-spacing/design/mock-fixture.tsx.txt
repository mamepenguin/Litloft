import type { ReactElement } from "react";
import { cardGridTemplate, useCardColumns } from "@/lib/cardGrid";

const VIDEO = { minWidthPx: 320, minWidth: "20rem", gapPx: 16 } as const;

function MeasuredPage({ children }: { children: ReactElement[] }): ReactElement {
  const { ref, columns } = useCardColumns(VIDEO);
  return (
    <ul ref={ref} className="grid" style={{ columnGap: 16, rowGap: 32, gridTemplateColumns: cardGridTemplate(columns, VIDEO) }}>
      {children}
    </ul>
  );
}

function video(i: number, progress = false): ShelfItem {
  return {
    id: `v${i}`,
    filename: `v${i}.mp4`,
    title: TITLES[i % TITLES.length],
    description: "",
    drive: "fixture",
    folder_path: "",
    file_type: "video",
    mime_type: "video/mp4",
    thumbnail_url: "",
    has_thumbnail: false,
    file_size: 1024,
    duration: 300 + i * 437,
    image_width: null,
    image_height: null,
    liked_at: null,
    is_favorite: false,
    tags: [],
    subtitles: [],
    deleted_at: null,
    missing_since: null,
    trust_tier: "verified",
    trust_reviewed_at: null,
    created_at: "2026-01-01T00:00:00",
    updated_at: "2026-01-01T00:00:00",
    ...(progress ? { watch_progress: { position: 40 + i * 30, duration: 300 + i * 437 } } : {}),
  } as ShelfItem;
}

const items = (from: number, n: number, progress = false) =>
  Array.from({ length: n }, (_, k) => video(from + k, progress));
const byline = (it: ShelfItem) => {
  const i = Number(it.id.slice(1));
  return `${CHANNELS[i % CHANNELS.length]} · ${DATES[i % DATES.length]}`;
};

export function VideoSpacingMock(): ReactElement {
  const variant = new URLSearchParams(location.search).get("variant") ?? "current";
  const page = new URLSearchParams(location.search).get("page") === "1";
  return (
    <NextIntlClientProvider locale="ja" messages={jaMessages}>
      <style>{PROPOSED + WIDE + THUMBS}</style>
      <div id="spacing-mock" data-variant={variant} className="bg-bg-primary px-4 pb-8 pt-4">
        {page ? (
          <section>
            <h1 className="mb-4 text-xl font-bold text-text-primary">購読の新着</h1>
            {variant === "current" ? (
              <ul className="grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-4 xl:grid-cols-5">
              {items(0, 15).map((it) => (
                <li key={it.id} className="min-w-0">
                  <VideoCard item={it} byline={byline(it)} />
                </li>
              ))}
              </ul>
            ) : (
              <MeasuredPage>
                {items(0, 16).map((it) => (
                  <li key={it.id} className="min-w-0">
                    <VideoCard item={it} byline={byline(it)} />
                  </li>
                ))}
              </MeasuredPage>
            )}
          </section>
        ) : (
          <div className="space-y-8">
            <ShelfBlock<ShelfItem> title="続きから" view={{ status: "loaded", items: items(3, 12, true) }} onRetry={() => {}} renderItem={(it) => <VideoCard item={it} />} />
            <ShelfBlock<ShelfItem> title="購読の新着" seeAllHref="#" view={{ status: "loaded", items: items(0, 12) }} onRetry={() => {}} renderItem={(it) => <VideoCard item={it} byline={byline(it)} />} />
            <ShelfBlock<ShelfItem> title="最近追加" seeAllHref="#" view={{ status: "loaded", items: items(5, 12) }} onRetry={() => {}} renderItem={(it) => <VideoCard item={it} />} />
          </div>
        )}
      </div>
    </NextIntlClientProvider>
  );
}
