"""Pass-through proxy to the host ollama that records the shape of each /api/chat request."""
import base64, hashlib, http.client, json, sys, time
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer

LOG = sys.argv[1]
UP = ("127.0.0.1", 11434)

class H(BaseHTTPRequestHandler):
    protocol_version = "HTTP/1.0"
    def _fwd(self, method):
        n = int(self.headers.get("Content-Length") or 0)
        body = self.rfile.read(n) if n else None
        if body and self.path.startswith("/api/chat"):
            try:
                d = json.loads(body)
                msgs = []
                for m in d.get("messages", []):
                    imgs = [hashlib.sha256(base64.b64decode(i)).hexdigest()[:16] + f":{len(base64.b64decode(i))}" for i in (m.get("images") or [])]
                    msgs.append({"role": m.get("role"), "content": m.get("content"), "images": imgs})
                with open(LOG, "a") as f:
                    f.write(json.dumps({"t": time.strftime("%F %T"), "path": self.path, "model": d.get("model"), "n_messages": len(msgs), "messages": msgs}, ensure_ascii=False) + "\n")
            except Exception as e:
                with open(LOG, "a") as f:
                    f.write(json.dumps({"error": str(e)}) + "\n")
        c = http.client.HTTPConnection(*UP, timeout=900)
        hdr = {k: v for k, v in self.headers.items() if k.lower() not in ("host", "connection")}
        c.request(method, self.path, body=body, headers=hdr)
        r = c.getresponse()
        self.send_response(r.status)
        for k, v in r.getheaders():
            if k.lower() not in ("transfer-encoding", "connection", "content-length"):
                self.send_header(k, v)
        self.send_header("Connection", "close")
        self.end_headers()
        while True:
            b = r.read1(65536)
            if not b:
                break
            self.wfile.write(b); self.wfile.flush()
    def do_POST(self): self._fwd("POST")
    def do_GET(self): self._fwd("GET")
    def log_message(self, *a): pass

ThreadingHTTPServer(("0.0.0.0", 11500), H).serve_forever()
